  <script>

    // === CONFIG ===

    const ADMIN_PASSWORD = 'admin123'; // changer avant usage réel

    const SLOTS = [

      '08:00 - 08:30',

      '09:00 - 09:30',

      '10:00 - 10:30',

      '11:00 - 11:30',

      '13:00 - 13:30',

      '14:00 - 14:30',

      '15:00 - 15:30',

    ];

    const CAPACITY = 5;

    const STORAGE_KEY = 'inscriptions_creneaux';

 

    // === util ===

    function loadData(){

      const raw = localStorage.getItem(STORAGE_KEY);

      if(!raw) return {};

      try{ return JSON.parse(raw);}catch(e){return {};}   

    }

    function saveData(d){ localStorage.setItem(STORAGE_KEY, JSON.stringify(d)); }

    function getSlotCount(slot){ const d=loadData(); return d[slot]?d[slot].length:0; }

    function addSignup(slot, entry){ const d=loadData(); d[slot]=d[slot]||[]; d[slot].push(entry); saveData(d);}   

    function clearAll(){ localStorage.removeItem(STORAGE_KEY); }

 

    // === populate selects ===

    const slotSelect = document.getElementById('slot');

    const filterSlot = document.getElementById('filter-slot');

    function refreshSlotOptions(){

      slotSelect.innerHTML='';

      filterSlot.innerHTML='';

      SLOTS.forEach(s=>{

        const count=getSlotCount(s);

        const opt=document.createElement('option');

        opt.value=s;

        opt.textContent=`${s} (${count}/${CAPACITY})`;

        if(count>=CAPACITY){ opt.disabled=true; }

        slotSelect.appendChild(opt);

 

        const fopt=document.createElement('option');

        fopt.value=s; fopt.textContent=s; filterSlot.appendChild(fopt);

      });

      // admin filter add "Tous"

      const allOpt = document.createElement('option');

      allOpt.value='__all'; allOpt.textContent='Tous';

      filterSlot.insertBefore(allOpt, filterSlot.firstChild);

    }

 

    function showStatus(msg, isError){

      const el=document.getElementById('status-message');

      el.textContent=msg;

      el.className=isError? 'error':'success';

    }

 

    document.getElementById('submit-btn').addEventListener('click',e=>{

      e.preventDefault();

      const name=document.getElementById('name').value.trim();

      const email=document.getElementById('email').value.trim();

      const slot=slotSelect.value;

      if(!name||!email){ showStatus('Nom et courriel requis.', true); return; }

      const currentCount=getSlotCount(slot);

      if(currentCount>=CAPACITY){ showStatus('Ce créneau est complet.', true); refreshSlotOptions(); return; }

      // simple duplicate check: same email in same slot

      const d=loadData();

      const existing=(d[slot]||[]).find(i=>i.email.toLowerCase()===email.toLowerCase());

      if(existing){ showStatus('Tu es déjà inscrit à ce créneau.', true); return; }

      const entry={name,email,when:new Date().toISOString()};

      addSignup(slot, entry);

      // show confirmation

      document.getElementById('user-form').classList.add('hidden');

      document.getElementById('confirmation').classList.remove('hidden');

      document.getElementById('conf-info').innerHTML=`<strong>${slot}</strong><br>Nom: ${name}<br>Courriel: ${email}`;

      showStatus('', false);

      refreshSlotOptions();

    });

    document.getElementById('new-signup').addEventListener('click',()=>{

      document.getElementById('user-form').classList.remove('hidden');

      document.getElementById('confirmation').classList.add('hidden');

      document.getElementById('name').value='';

      document.getElementById('email').value='';

    });

 

    // admin login toggle

    document.getElementById('admin-toggle').addEventListener('click',()=>{

      const panel=document.getElementById('admin-panel');

      if(!panel.classList.contains('hidden')){ // logout

        panel.classList.add('hidden');

        document.getElementById('admin-login').classList.remove('hidden');

        document.getElementById('admin-toggle').textContent='Se connecter';

      }

    });

    document.getElementById('login-btn').addEventListener('click',()=>{

      const pw=document.getElementById('admin-password').value;

      const err=document.getElementById('admin-error');

      if(pw!==ADMIN_PASSWORD){ err.textContent='Mot de passe incorrect.'; return; }

      err.textContent='';

      document.getElementById('admin-login').classList.add('hidden');

      document.getElementById('admin-panel').classList.remove('hidden');

      document.getElementById('admin-toggle').textContent='Se déconnecter';

      refreshAdminView();

    });

 

    function buildAdminList(filter){

      const d=loadData();

      let slotsToShow = Object.keys(d);

      if(filter && filter!=='__all'){ slotsToShow=[filter]; }

      let html = '';

      let total=0;

      slotsToShow.forEach(slot=>{

        const entries=d[slot]||[];

        total += entries.length;

        html += `<div style="margin-bottom:1rem;"><div class="pill"><strong>${slot}</strong> — ${entries.length}/${CAPACITY}</div>`;

        if(entries.length){

          html += '<table class="table"><thead><tr><th>Nom</th><th>Courriel</th><th>Inscrit le</th></tr></thead><tbody>';

          entries.forEach(e=>{

            html += `<tr><td>${e.name}</td><td>${e.email}</td><td>${new Date(e.when).toLocaleString('fr-CA')}</td></tr>`;

          });

          html += '</tbody></table>';

        } else {

          html += '<div style="margin-top:4px;">Aucun inscrits.</div>';

        }

        html += '</div>';

      });

      document.getElementById('list-container').innerHTML=html||'<div>Aucune inscription.</div>';

      document.getElementById('admin-summary').innerHTML=`<strong>Total inscriptions:</strong> ${total}`;

    }

 

    function refreshAdminView(){

      refreshSlotOptions();

      buildAdminList(filterSlot.value);

    }

    filterSlot.addEventListener('change',()=>{ buildAdminList(filterSlot.value); });

 

    document.getElementById('export-btn').addEventListener('click',()=>{

      const d=loadData();

      let csv='Créneau,Nom,Courriel,Inscrit le\n';

      Object.entries(d).forEach(([slot,arr])=>{

        arr.forEach(e=>{

          csv += `"${slot}","${e.name}","${e.email}","${new Date(e.when).toLocaleString('fr-CA')}"\n`;

        });

      });

      const blob=new Blob([csv],{type:'text/csv'});

      const url=URL.createObjectURL(blob);

      const a=document.createElement('a');

      a.href=url; a.download='inscriptions.csv';

      document.body.appendChild(a); a.click();

      document.body.removeChild(a);

    });

    document.getElementById('reset-btn').addEventListener('click',()=>{

      if(confirm('Tout réinitialiser ? Les données seront perdues.')){ clearAll(); refreshAdminView(); }

    });

 

    // initial render

    refreshSlotOptions();

  </script>
